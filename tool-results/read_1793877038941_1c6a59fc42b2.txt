    55→    55→  return canvas;

    92→    92→async function runMerge(
    93→    93→  entries: { id: string; file: File }[],
    94→    94→  cb: RunCallbacks
    95→    95→): Promise<EngineResult> {
    96→    96→  const { PDFDocument } = await getPdfLib();
    97→    97→  const out = await PDFDocument.create();
    98→    98→  for (let i = 0; i < entries.length; i++) {
    99→    99→    const bytes = await fileToBytes(entries[i].file);
   100→   100→    let src;
   101→   101→    try {
   102→   102→      src = await loadWithPdfLib(bytes);
   103→   103→    } catch {
   104→   104→      throw new Error(
   105→   105→        `"${entries[i].file.name}" could not be read. If it is password-protected, unlock it first.`
   106→   106→      );
   107→   107→    }
   108→   108→    const pages = await out.copyPages(src, src.getPageIndices());
   109→   109→    for (const p of pages) out.addPage(p);
   110→   110→    cb.onFileProgress?.(entries[i].id, 1);
   111→   111→    cb.onOverall?.((i + 1) / entries.length);
   112→   112→    await tick();
   113→   113→  }
   114→   114→  const blob = new Blob([await out.save()], { type: "application/pdf" });
   115→   115→  return {
   116→   116→    files: [{ name: "merged.pdf", blob, size: blob.size, kind: "PDF" }],
   117→   117→    note: `${entries.length} PDFs combined`,
   118→   118→  };
   119→   119→}

   123→   123→export interface SplitOptions {
   124→   124→  mode: "ranges" | "fixed" | "extract";

   129→   129→
   130→   130→function parseRangesStrict(spec: string, total: number): number[][] {
   131→   131→  const groups: number[][] = [];
   132→   132→  const parts = spec
   133→   133→    .split(",")
   134→   134→    .map((s) => s.trim())
   135→   135→    .filter(Boolean);
   136→   136→  if (!parts.length) throw new Error("Enter at least one page range");
   137→   137→  for (const part of parts) {
   138→   138→    const m = part.match(/^(\d+)\s*-\s*(\d+)$/);
   139→   139→    if (m) {
   140→   140→      let a = parseInt(m[1], 10);
   141→   141→      let b = parseInt(m[2], 10);
   142→   142→      if (a > b) [a, b] = [b, a];
   143→   143→      if (a < 1 || b > total)
   144→   144→        throw new Error(`Range "${part}" is outside 1–${total}`);
   145→   145→      const g: number[] = [];
   146→   146→      for (let i = a; i <= b; i++) g.push(i - 1);
   147→   147→      groups.push(g);
   148→   148→    } else if (/^\d+$/.test(part)) {
   149→   149→      const n = parseInt(part, 10);
   150→   150→      if (n < 1 || n > total)
   151→   151→        throw new Error(`Page ${n} is outside 1–${total}`);
   152→   152→      groups.push([n - 1]);
   153→   153→    } else {
   154→   154→      throw new Error(`"${part}" is not a valid page or range`);
   155→   155→    }
   156→   156→  }
   157→   157→  return groups;
   158→   158→}

   160→   160→async function runSplit(
   161→   161→  entries: { id: string; file: File }[],
   162→   162→  opts: SplitOptions,
   163→   163→  cb: RunCallbacks
   164→   164→): Promise<EngineResult> {
   165→   165→  const { PDFDocument } = await getPdfLib();
   166→   166→  const bytes = await fileToBytes(entries[0].file);
   167→   167→  const src = await loadWithPdfLib(bytes);
   168→   168→  const total = src.getPageCount();
   169→   169→  const base = entries[0].file.name.replace(/\.pdf$/i, "");
   170→   170→
   171→   171→  let groups: number[][] = [];
   172→   172→  if (opts.mode === "ranges") {
   173→   173→    groups = parseRangesStrict(opts.ranges, total);
   174→   174→  } else if (opts.mode === "fixed") {
   175→   175→    const n = Math.max(1, Math.floor(opts.everyN || 1));
   176→   176→    for (let i = 0; i < total; i += n) {
   177→   177→      const g: number[] = [];
   178→   178→      for (let j = i; j < Math.min(i + n, total); j++) g.push(j);
   179→   179→      groups.push(g);
   180→   180→    }
   181→   181→  } else {
   182→   182→    const sel = [...new Set(opts.selectedPages)].sort((a, b) => a - b);
   183→   183→    if (!sel.length) throw new Error("Select at least one page below");
   184→   184→    groups = [sel.map((p) => p - 1)];
   185→   185→  }
   186→   186→
   187→   187→  const outFiles: EngineResult["files"] = [];
   188→   188→  const many = groups.length > 1;
   189→   189→  const pad = String(groups.length).length;
   190→   190→
   191→   191→  for (let gi = 0; gi < groups.length; gi++) {
   192→   192→    const out = await PDFDocument.create();
   193→   193→    const pages = await out.copyPages(src, groups[gi]);
   194→   194→    for (const p of pages) out.addPage(p);
   195→   195→    const blob = new Blob([await out.save()], { type: "application/pdf" });
   196→   196→    const label = many
   197→   197→      ? `${base}-part${padNum(gi + 1, pad)}.pdf`
   198→   198→      : `${base}-${groups[gi].length}pages.pdf`;
   199→   199→    outFiles.push({ name: label, blob, size: blob.size, kind: "PDF" });
   200→   200→    cb.onOverall?.((gi + 1) / groups.length);
   201→   201→    await tick();
   202→   202→  }
   203→   203→  return { files: outFiles, note: `${groups.length} file(s) created` };
   204→   204→}

   207→   207→
   208→   208→export interface CompressOptions {
   209→   209→  level: "low" | "medium" | "high";
   210→   210→}
   211→   211→
   212→   212→async function runCompress(
   213→   213→  entries: { id: string; file: File }[],
   214→   214→  opts: CompressOptions,
   215→   215→  cb: RunCallbacks
   216→   216→): Promise<EngineResult> {
   217→   217→  const settings =
   218→   218→    opts.level === "high"
   219→   219→      ? { scale: 2.0, quality: 0.92 }
   220→   220→      : opts.level === "medium"
   221→   221→        ? { scale: 1.5, quality: 0.8 }
   222→   222→        : { scale: 1.1, quality: 0.62 };
   223→   223→
   224→   224→  const outFiles: EngineResult["files"] = [];
   225→   225→  let totalIn = 0;
   226→   226→  let totalOut = 0;
   227→   227→
   228→   228→  for (let i = 0; i < entries.length; i++) {
   229→   229→    const { id, file } = entries[i];
   230→   230→    totalIn += file.size;
   231→   231→    const bytes = await fileToBytes(file);
   232→   232→    cb.onStatus?.(`Optimizing ${file.name}`);
   233→   233→
   234→   234→    // Strategy A — structural re-save (keeps text sharp)
   235→   235→    let aSize = Infinity;
   236→   236→    let aBytes: Uint8Array | null = null;
   237→   237→    try {
   238→   238→      const docA = await loadWithPdfLib(bytes);
   239→   239→      aBytes = await docA.save({ useObjectStreams: true });
   240→   240→      aSize = aBytes.length;
   241→   241→    } catch {
   242→   242→      /* ignore */
   243→   243→    }
   244→   244→
   245→   245→    // Strategy B — rasterize pages (great for scans)
   246→   246→    let bSize = Infinity;
   247→   247→    let bBlob: Blob | null = null;
   248→   248→    try {
   249→   249→      const doc = await openWithPdfjs(bytes);
   250→   250→      const { PDFDocument } = await getPdfLib();
   251→   251→      const out = await PDFDocument.create();
   252→   252→      const n = doc.numPages;
   253→   253→      for (let p = 1; p <= n; p++) {
   254→   254→        const canvas = await renderPageToCanvas(doc, p, settings.scale);
   255→   255→        const jpg = await canvasToBlob(canvas, "image/jpeg", settings.quality);
   256→   256→        const img = await out.embedJpg(new Uint8Array(await jpg.arrayBuffer()));
   257→   257→        const page = out.addPage([img.width, img.height]);
   258→   258→        page.drawImage(img, {
   259→   259→          x: 0,
   260→   260→          y: 0,
   261→   261→          width: img.width,
   262→   262→          height: img.height,
   263→   263→        });
   264→   264→        cb.onFileProgress?.(id, (p / n) * 0.9);
   265→   265→        await tick();
   266→   266→      }
   267→   267→      const saved = await out.save({ useObjectStreams: true });
   268→   268→      bBlob = new Blob([saved], { type: "application/pdf" });
   269→   269→      bSize = saved.length;
   270→   270→    } catch {
   271→   271→      /* ignore */
   272→   272→    }
   273→   273→
   274→   274→    let outBlob: Blob;
   275→   275→    if (aSize <= bSize && aBytes) {
   276→   276→      outBlob = new Blob([aBytes], { type: "application/pdf" });
   277→   277→    } else if (bBlob) {
   278→   278→      outBlob = bBlob;
   279→   279→    } else if (aBytes) {
   280→   280→      outBlob = new Blob([aBytes], { type: "application/pdf" });
   281→   281→    } else {
   282→   282→      throw new Error(`"${file.name}" could not be compressed`);
   283→   283→    }
   284→   284→    totalOut += outBlob.size;
   285→   285→    outFiles.push({
   286→   286→      name: `${file.name.replace(/\.pdf$/i, "")}-compressed.pdf`,
   287→   287→      blob: outBlob,
   288→   288→      size: outBlob.size,
   289→   289→      kind: "PDF",
   290→   290→    });
   291→   291→    cb.onFileProgress?.(id, 1);
   292→   292→    cb.onOverall?.((i + 1) / entries.length);
   293→   293→  }
   294→   294→
   295→   295→  const note =
   296→   296→    totalOut < totalIn
   297→   297→      ? `Saved ${Math.max(1, Math.round((1 - totalOut / totalIn) * 100))}% of size`
   298→   298→      : "Already optimized — kept the smaller re-save";
   299→   299→  return { files: outFiles, note };
   300→   300→}

   304→   304→export interface OrganizeOptions {
   305→   305→  /** page order (1-based original numbers) after user edits */
   306→   306→  order: number[];
   307→   307→  /** extra rotation per original page number */
   308→   308→  rotations: Record<number, number>;
   309→   309→}
   310→   310→
   311→   311→async function runOrganize(
   312→   312→  entries: { id: string; file: File }[],
   313→   313→  opts: OrganizeOptions,
   314→   314→  cb: RunCallbacks
   315→   315→): Promise<EngineResult> {
   316→   316→  const { PDFDocument, degrees } = await getPdfLib();
   317→   317→  const bytes = await fileToBytes(entries[0].file);
   318→   318→  const src = await loadWithPdfLib(bytes);
   319→   319→  if (!opts.order.length) throw new Error("Keep at least one page");
   320→   320→
   321→   321→  const out = await PDFDocument.create();
   322→   322→  const total = opts.order.length;
   323→   323→  for (let i = 0; i < total; i++) {
   324→   324→    const original1based = opts.order[i];
   325→   325→    const [copied] = await out.copyPages(src, [original1based - 1]);
   326→   326→    const extra = opts.rotations[original1based] ?? 0;
   327→   327→    if (extra) {
   328→   328→      const cur = copied.getRotation().angle;
   329→   329→      copied.setRotation(degrees((cur + extra) % 360));
   330→   330→    }
   331→   331→    out.addPage(copied);
   332→   332→    cb.onOverall?.((i + 1) / total);
   333→   333→    if (i % 4 === 0) await tick();
   334→   334→  }
   335→   335→  const blob = new Blob([await out.save()], { type: "application/pdf" });
   336→   336→  return {
   337→   337→    files: [
   338→   338→      {
   339→   339→        name: `${entries[0].file.name.replace(/\.pdf$/i, "")}-organized.pdf`,
   340→   340→        blob,
   341→   341→        size: blob.size,
   342→   342→        kind: "PDF",
   343→   343→      },
   344→   344→    ],
   345→   345→    note: `${total} page${total === 1 ? "" : "s"} kept`,
   346→   346→  };
   347→   347→}

   424→   424→/* ---------------- JPG → PDF ---------------- */
   425→   425→
   426→   426→export interface JpgToPdfOptions {
   427→   427→  pageSize: "fit" | "a4" | "letter";
   428→   428→  margin: number; // pt
   429→   429→  orientation: "auto" | "portrait" | "landscape";
   430→   430→}
   431→   431→
   432→   432→async function runJpgToPdf(
   433→   433→  entries: { id: string; file: File }[],
   434→   434→  opts: JpgToPdfOptions,
   435→   435→  cb: RunCallbacks
   436→   436→): Promise<EngineResult> {
   437→   437→  const { PDFDocument } = await getPdfLib();
   438→   438→  const out = await PDFDocument.create();
   439→   439→  const A4 = { w: 595.28, h: 841.89 };
   440→   440→  const LETTER = { w: 612, h: 792 };
   441→   441→
   442→   442→  for (let i = 0; i < entries.length; i++) {
   443→   443→    const bytes = new Uint8Array(await entries[i].file.arrayBuffer());
   444→   444→    let img;
   445→   445→    try {
   446→   446→      img = await out.embedJpg(bytes);
   447→   447→    } catch {
   448→   448→      img = await out.embedPng(bytes);
   449→   449→    }
   450→   450→
   451→   451→    let pageW: number;
   452→   452→    let pageH: number;
   453→   453→    if (opts.pageSize === "fit") {
   454→   454→      pageW = img.width + opts.margin * 2;
   455→   455→      pageH = img.height + opts.margin * 2;
   456→   456→    } else {
   457→   457→      const std = opts.pageSize === "a4" ? A4 : LETTER;
   458→   458→      const landscape =
   459→   459→        opts.orientation === "landscape" ||
   460→   460→        (opts.orientation === "auto" && img.width > img.height);
   461→   461→      pageW = landscape ? std.h : std.w;
   462→   462→      pageH = landscape ? std.w : std.h;
   463→   463→    }
   464→   464→    const page = out.addPage([pageW, pageH]);
   465→   465→    const maxW = pageW - opts.margin * 2;
   466→   466→    const maxH = pageH - opts.margin * 2;
   467→   467→    const ratio = Math.min(maxW / img.width, maxH / img.height);
   468→   468→    const w = img.width * ratio;
   469→   469→    const h = img.height * ratio;
   470→   470→    page.drawImage(img, {
   471→   471→      x: (pageW - w) / 2,
   472→   472→      y: (pageH - h) / 2,
   473→   473→      width: w,
   474→   474→      height: h,
   475→   475→    });
   476→   476→    cb.onFileProgress?.(entries[i].id, 1);

   498→   498→
   499→   499→async function runPdfToWord(
   500→   500→  entries: { id: string; file: File }[],
   501→   501→  _opts: Record<string, never>,
   502→   502→  cb: RunCallbacks
   503→   503→): Promise<EngineResult> {
   504→   504→  const { Document, Packer, Paragraph, TextRun, HeadingLevel } = await import(
   505→   505→    "docx"
   506→   506→  );
   507→   507→  const outFiles: EngineResult["files"] = [];
   508→   508→
   509→   509→  for (let i = 0; i < entries.length; i++) {
   510→   510→    const bytes = await fileToBytes(entries[i].file);
   511→   511→    const doc = await openWithPdfjs(bytes);
   512→   512→    const n = doc.numPages;
   513→   513→    const children: any[] = [];
   514→   514→
   515→   515→    for (let p = 1; p <= n; p++) {
   516→   516→      const page = await doc.getPage(p);
   517→   517→      const tc = await page.getTextContent();
   518→   518→      const lines: {
   519→   519→        y: number;
   520→   520→        items: { x: number; str: string; bold: boolean; size: number }[];
   521→   521→      }[] = [];
   522→   522→      for (const item of tc.items as any[]) {
   523→   523→        if (!item.str || !item.str.trim()) continue;
   524→   524→        const x = item.transform[4];
   525→   525→        const y = item.transform[5];
   526→   526→        const size = Math.hypot(item.transform[0], item.transform[1]) || 12;
   527→   527→        const bold = /bold|black|heavy/i.test(String(item.fontName || ""));
   528→   528→        let line = lines.find((l) => Math.abs(l.y - y) < size * 0.6);
   529→   529→        if (!line) {
   530→   530→          line = { y, items: [] };
   531→   531→          lines.push(line);
   532→   532→        }
   533→   533→        line.items.push({ x, str: item.str, bold, size });
   534→   534→      }
   535→   535→      lines.sort((a, b) => b.y - a.y);
   536→   536→
   537→   537→      if (p > 1) children.push(new Paragraph({ text: "" }));
   538→   538→      for (const line of lines) {
   539→   539→        line.items.sort((a, b) => a.x - b.x);
   540→   540→        let text = "";
   541→   541→        let prevEnd: number | null = null;
   542→   542→        for (const it of line.items) {
   543→   543→          if (prevEnd !== null) {
   544→   544→            const gap = it.x - prevEnd;
   545→   545→            if (gap > 1.5) text += gap > 12 ? "  " : " ";
   546→   546→          }
   547→   547→          text += it.str;
   548→   548→          prevEnd = it.x + it.str.length * it.size * 0.5;
   549→   549→        }
   550→   550→        if (!text.trim()) continue;
   551→   551→        const maxSize = Math.max(...line.items.map((s) => s.size));
   552→   552→        const allBold = line.items.every((s) => s.bold);
   553→   553→        if (allBold && maxSize > 13) {
   554→   554→          children.push(
   555→   555→            new Paragraph({
   556→   556→              heading:
   557→   557→                maxSize > 17 ? HeadingLevel.HEADING_1 : HeadingLevel.HEADING_2,
   558→   558→              children: [new TextRun({ text, bold: true })],
   559→   559→            })
   560→   560→          );
   561→   561→        } else {
   562→   562→          children.push(
   563→   563→            new Paragraph({
   564→   564→              children: [
   565→   565→                new TextRun({
   566→   566→                  text,
   567→   567→                  size: Math.max(16, Math.round(maxSize * 2 * 0.75)),
   568→   568→                  bold: allBold,
   569→   569→                }),
   570→   570→              ],
   571→   571→            })
   572→   572→          );
   573→   573→        }
   574→   574→      }
   575→   575→      cb.onStatus?.(`Extracting page ${p}/${n}`);
   576→   576→      cb.onFileProgress?.(entries[i].id, p / n);
   577→   577→      cb.onOverall?.(((i + p / n) / entries.length) * 0.92);
   578→   578→      await tick();
   579→   579→    }
   580→   580→
   581→   581→    const docxDoc = new Document({ sections: [{ children }] });
   582→   582→    const blob = await Packer.toBlob(docxDoc);
   583→   583→    outFiles.push({
   584→   584→      name: `${entries[i].file.name.replace(/\.pdf$/i, "")}.docx`,
   585→   585→      blob,
   586→   586→      size: blob.size,
   587→   587→      kind: "DOCX",
   588→   588→    });
   589→   589→    cb.onFileProgress?.(entries[i].id, 1);
   590→   590→  }
   591→   591→  cb.onOverall?.(1);
   592→   592→  return {
   593→   593→    files: outFiles,
   594→   594→    note: "Text layer extracted — scanned PDFs have no text layer",
   595→   595→  };
   596→   596→}

   663→   663→

   666→   666→    type: "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
   667→   667→  });
   668→   668→  const name =
   669→   669→    entries.length === 1
   670→   670→      ? `${entries[0].file.name.replace(/\.pdf$/i, "")}.xlsx`
   671→   671→      : "tables.xlsx";
   672→   672→  cb.onOverall?.(1);
   673→   673→  return {
   674→   674→    files: [{ name, blob, size: blob.size, kind: "XLSX" }],

   676→   676→  };

   746→   746→async function runWordToPdf(
   747→   747→  entries: { id: string; file: File }[],
   748→   748→  opts: WordToPdfOptions,
   749→   749→  cb: RunCallbacks
   750→   750→): Promise<EngineResult> {
   751→   751→  const mammoth = await import("mammoth");
   752→   752→  const { jsPDF } = await import("jspdf");
   753→   753→  const outFiles: EngineResult["files"] = [];
   754→   754→  const pageW = opts.pageSize === "a4" ? 595.28 : 612;
   755→   755→  const pageH = opts.pageSize === "a4" ? 841.89 : 792;
   756→   756→  const M = 56;
   757→   757→
   758→   758→  for (let i = 0; i < entries.length; i++) {
   759→   759→    const buf = await entries[i].file.arrayBuffer();
   760→   760→    const { value: html } = await mammoth.convertToHtml({ arrayBuffer: buf });
   761→   761→    const dom = new DOMParser().parseFromString(html, "text/html");
   762→   762→    const pdf = new jsPDF({ unit: "pt", format: [pageW, pageH] });
   763→   763→    let y = M;
   764→   764→
   765→   765→    const writeBlock = (
   766→   766→      text: string,
   767→   767→      size: number,
   768→   768→      bold: boolean,
   769→   769→      gapAfter: number
   770→   770→    ) => {
   771→   771→      const clean = text.replace(/\s+/g, " ").trim();
   772→   772→      if (!clean) return;
   773→   773→      pdf.setFont("helvetica", bold ? "bold" : "normal");
   774→   774→      pdf.setFontSize(size);
   775→   775→      const lines = pdf.splitTextToSize(clean, pageW - M * 2) as string[];
   776→   776→      for (const ln of lines) {
   777→   777→        if (y > pageH - M) {
   778→   778→          pdf.addPage();
   779→   779→          y = M;
   780→   780→        }
   781→   781→        pdf.text(ln, M, y);
   782→   782→        y += size * 1.4;
   783→   783→      }
   784→   784→      y += gapAfter;
   785→   785→    };
   786→   786→
   787→   787→    const walk = (node: Element) => {
   788→   788→      for (const el of Array.from(node.children)) {
   789→   789→        const tag = el.tagName.toLowerCase();
   790→   790→        const text = el.textContent ?? "";
   791→   791→        if (/^h[1-6]$/.test(tag)) {
   792→   792→          const size = [20, 17, 15, 13.5, 12.5, 12][parseInt(tag[1], 10) - 1];
   793→   793→          writeBlock(text, size, true, 6);
   794→   794→        } else if (tag === "li") {
   795→   795→          writeBlock(`•  ${text}`, 11, false, 2);
   796→   796→        } else if (tag === "table") {
   797→   797→          const rows = Array.from(el.querySelectorAll("tr"));
   798→   798→          for (const tr of rows) {
   799→   799→            const cells = Array.from(tr.querySelectorAll("th,td")).map(
   800→   800→              (c) => (c.textContent ?? "").trim()
   801→   801→            );
   802→   802→            writeBlock(
   803→   803→              cells.join("    |    "),
   804→   804→              10,
   805→   805→              tr.querySelector("th") !== null,
   806→   806→              2
   807→   807→            );
   808→   808→          }
   809→   809→          y += 6;
   810→   810→        } else if (el.children.length) {
   811→   811→          walk(el);
   812→   812→        } else {
   813→   813→          writeBlock(text, 11, false, 8);
   814→   814→        }
   815→   815→      }
   816→   816→    };
   817→   817→    walk(dom.body);
   818→   818→
   819→   819→    const blob = pdf.output("blob");
   820→   820→    outFiles.push({
   821→   821→      name: `${entries[i].file.name.replace(/\.docx$/i, "")}.pdf`,
   822→   822→      blob,
   823→   823→      size: blob.size,
   824→   824→      kind: "PDF",
   825→   825→    });
   826→   826→    cb.onFileProgress?.(entries[i].id, 1);
   827→   827→    cb.onOverall?.((i + 1) / entries.length);
   828→   828→  }
   829→   829→  return {
   830→   830→    files: outFiles,
   831→   831→    note: "Text & structure kept — complex layout simplified",
   832→   832→  };
   833→   833→}

   835→   835→/* ---------------- WATERMARK ---------------- */
   836→   836→
   837→   837→export interface WatermarkOptions {
   838→   838→  text: string;
   839→   839→  fontSize: number;
   840→   840→  color: string; // hex
   841→   841→  opacity: number; // 0..1
   842→   842→  rotation: number;
   843→   843→  position: "center" | "diagonal" | "tile" | "footer" | "header";
   844→   844→  bold: boolean;
   845→   845→}
   846→   846→
   847→   847→function hexToRgb01(hex: string): [number, number, number] {
   848→   848→  const m = hex.replace("#", "");
   849→   849→  const r = parseInt(m.slice(0, 2), 16) / 255;
   850→   850→  const g = parseInt(m.slice(2, 4), 16) / 255;
   851→   851→  const b = parseInt(m.slice(4, 6), 16) / 255;
   852→   852→  return [r, g, b];
   853→   853→}
   854→   854→
   855→   855→async function runWatermark(
   856→   856→  entries: { id: string; file: File }[],
   857→   857→  opts: WatermarkOptions,
   858→   858→  cb: RunCallbacks
   859→   859→): Promise<EngineResult> {
   860→   860→  const { PDFDocument, degrees, rgb, StandardFonts } = await getPdfLib();
   861→   861→  if (!opts.text.trim()) throw new Error("Enter watermark text");
   862→   862→  const holder = await PDFDocument.create();
   863→   863→  const font = opts.bold
   864→   864→    ? await holder.embedFont(StandardFonts.HelveticaBold)
   865→   865→    : await holder.embedFont(StandardFonts.Helvetica);
   866→   866→  const [r, g, b] = hexToRgb01(opts.color);
   867→   867→
   868→   868→  const outFiles: EngineResult["files"] = [];
   869→   869→  for (let i = 0; i < entries.length; i++) {
   870→   870→    const bytes = await fileToBytes(entries[i].file);
   871→   871→    const doc = await loadWithPdfLib(bytes);
   872→   872→    const pages = doc.getPages();
   873→   873→    for (const page of pages) {
   874→   874→      const { width, height } = page.getSize();
   875→   875→      const tw = font.widthOfTextAtSize(opts.text, opts.fontSize);
   876→   876→      const draw = (x: number, y: number, rot: number) => {
   877→   877→        page.drawText(opts.text, {
   878→   878→          x,
   879→   879→          y,
   880→   880→          size: opts.fontSize,
   881→   881→          font,
   882→   882→          color: rgb(r, g, b),
   883→   883→          opacity: opts.opacity,
   884→   884→          rotate: degrees(rot),
   885→   885→        });
   886→   886→      };
   887→   887→      switch (opts.position) {
   888→   888→        case "center":
   889→   889→          draw((width - tw) / 2, height / 2 - opts.fontSize / 2, 0);
   890→   890→          break;
   891→   891→        case "diagonal":
   892→   892→          draw((width - tw) / 2 - width * 0.1, height * 0.35, opts.rotation);
   893→   893→          break;
   894→   894→        case "tile": {
   895→   895→          const stepX = Math.max(40, tw + width * 0.08);
   896→   896→          const stepY = Math.max(30, opts.fontSize * 5);
   897→   897→          for (let y = 0; y < height + stepY; y += stepY) {
   898→   898→            for (let x = -tw; x < width + stepX; x += stepX) {
   899→   899→              draw(x, y, opts.rotation);
   900→   900→            }
   901→   901→          }
   902→   902→          break;
   903→   903→        }
   904→   904→        case "footer":
   905→   905→          draw((width - tw) / 2, opts.fontSize * 1.2, 0);
   906→   906→          break;
   907→   907→        case "header":
   908→   908→          draw((width - tw) / 2, height - opts.fontSize * 2, 0);
   909→   909→          break;
   910→   910→      }
   911→   911→    }
   912→   912→    const blob = new Blob([await doc.save()], { type: "application/pdf" });
   913→   913→    outFiles.push({
   914→   914→      name: `${entries[i].file.name.replace(/\.pdf$/i, "")}-watermarked.pdf`,
   915→   915→      blob,
   916→   916→      size: blob.size,
   917→   917→      kind: "PDF",
   918→   918→    });
   919→   919→    cb.onFileProgress?.(entries[i].id, 1);
   920→   920→    cb.onOverall?.((i + 1) / entries.length);
   921→   921→    await tick();
   922→   922→  }
   923→   923→  return { files: outFiles, note: `"${opts.text}" stamped on every page` };
   924→   924→}

   928→   928→export interface PageNumberOptions {
   929→   929→  format: "n" | "page_n" | "n_of_total";
   930→   930→  position:
   931→   931→    | "bottom-center"
   932→   932→    | "bottom-right"
   933→   933→    | "bottom-left"
   934→   934→    | "top-center"
   935→   935→    | "top-right"
   936→   936→    | "top-left";
   937→   937→  startAt: number;
   938→   938→  fontSize: number;
   939→   939→  skipFirst: boolean;
   940→   940→}
   941→   941→
   942→   942→async function runPageNumbers(
   943→   943→  entries: { id: string; file: File }[],
   944→   944→  opts: PageNumberOptions,
   945→   945→  cb: RunCallbacks
   946→   946→): Promise<EngineResult> {
   947→   947→  const { PDFDocument, StandardFonts } = await getPdfLib();
   948→   948→  const outFiles: EngineResult["files"] = [];
   949→   949→
   950→   950→  for (let i = 0; i < entries.length; i++) {
   951→   951→    const bytes = await fileToBytes(entries[i].file);
   952→   952→    const doc = await loadWithPdfLib(bytes);
   953→   953→    const font = await doc.embedFont(StandardFonts.Helvetica);
   954→   954→    const pages = doc.getPages();
   955→   955→    const total = pages.length;
   956→   956→    const showFrom = opts.skipFirst ? 2 : 1;
   957→   957→
   958→   958→    for (let p = 1; p <= total; p++) {
   959→   959→      if (p < showFrom) continue;
   960→   960→      const num = opts.startAt + p - 1;
   961→   961→      const label =
   962→   962→        opts.format === "n"
   963→   963→          ? `${num}`
   964→   964→          : opts.format === "page_n"
   965→   965→            ? `Page ${num}`
   966→   966→            : `${num} / ${opts.startAt + total - 1}`;
   967→   967→      const page = pages[p - 1];
   968→   968→      const { width, height } = page.getSize();
   969→   969→      const tw = font.widthOfTextAtSize(label, opts.fontSize);
   970→   970→      const M = 28;
   971→   971→      let x = (width - tw) / 2;
   972→   972→      let y = M;
   973→   973→      if (opts.position.endsWith("right")) x = width - M - tw;
   974→   974→      if (opts.position.endsWith("left")) x = M;
   975→   975→      if (opts.position.startsWith("top")) y = height - M - opts.fontSize;
   976→   976→      page.drawText(label, { x, y, size: opts.fontSize, font });
   977→   977→    }
   978→   978→    const blob = new Blob([await doc.save()], { type: "application/pdf" });
   979→   979→    outFiles.push({
   980→   980→      name: `${entries[i].file.name.replace(/\.pdf$/i, "")}-numbered.pdf`,
   981→   981→      blob,
   982→   982→      size: blob.size,
   983→   983→      kind: "PDF",
   984→   984→    });
   985→   985→    cb.onFileProgress?.(entries[i].id, 1);
   986→   986→    cb.onOverall?.((i + 1) / entries.length);
   987→   987→    await tick();
   988→   988→  }
   989→   989→  return { files: outFiles, note: "Numbers placed" };
   990→   990→}

   994→   994→export interface CropOptions {
   995→   995→  top: number; // %
   996→   996→  right: number;
   997→   997→  bottom: number;
   998→   998→  left: number;
   999→   999→}
  1000→  1000→
  1001→  1001→async function runCrop(
  1002→  1002→  entries: { id: string; file: File }[],
  1003→  1003→  opts: CropOptions,
  1004→  1004→  cb: RunCallbacks
  1005→  1005→): Promise<EngineResult> {
  1006→  1006→  const { PDFDocument } = await getPdfLib();
  1007→  1007→  const outFiles: EngineResult["files"] = [];
  1008→  1008→  const clamp = (v: number) => Math.max(0, Math.min(45, v)) / 100;
  1009→  1009→  const t = clamp(opts.top);
  1010→  1010→  const rr = clamp(opts.right);
  1011→  1011→  const b = clamp(opts.bottom);
  1012→  1012→  const l = clamp(opts.left);
  1013→  1013→
  1014→  1014→  for (let i = 0; i < entries.length; i++) {
  1015→  1015→    const bytes = await fileToBytes(entries[i].file);
  1016→  1016→    const doc = await loadWithPdfLib(bytes);
  1017→  1017→    for (const page of doc.getPages()) {
  1018→  1018→      const { width, height } = page.getSize();
  1019→  1019→      const newW = width * (1 - rr - l);
  1020→  1020→      const newH = height * (1 - t - b);
  1021→  1021→      page.setCropBox(width * l, height * b, newW, newH);
  1022→  1022→    }
  1023→  1023→    const blob = new Blob([await doc.save()], { type: "application/pdf" });
  1024→  1024→    outFiles.push({
  1025→  1025→      name: `${entries[i].file.name.replace(/\.pdf$/i, "")}-cropped.pdf`,
  1026→  1026→      blob,
  1027→  1027→      size: blob.size,
  1028→  1028→      kind: "PDF",
  1029→  1029→    });
  1030→  1030→    cb.onFileProgress?.(entries[i].id, 1);
  1031→  1031→    cb.onOverall?.((i + 1) / entries.length);
  1032→  1032→    await tick();
  1033→  1033→  }
  1034→  1034→  return { files: outFiles, note: "Crop box applied to all pages" };
  1035→  1035→}

  1046→  1046→  opts: ProtectOptions,
  1047→  1047→  cb: RunCallbacks
  1048→  1048→): Promise<EngineResult> {
  1049→  1049→  if (!opts.password || opts.password.length < 4)
  1050→  1050→    throw new Error("Use a password of at least 4 characters");
  1051→  1051→  if (opts.password !== opts.confirm) throw new Error("Passwords do not match");
  1052→  1052→

  1059→  1059→      ignoreEncryption: true,

  1071→  1071→        copying: false,

  1074→  1074→        contentAccessibility: true,

  1086→  1086→    cb.onFileProgress?.(entries[i].id, 1);
  1087→  1087→    cb.onOverall?.((i + 1) / entries.length);

  1089→  1089→  }

  1092→  1092→    note: "Password set — keep it safe, it cannot be recovered",

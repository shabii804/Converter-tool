   352→function entries(files: StoreFile[]) {
   353→  return files.map((f) => ({ id: f.id, file: f.file }));

   355→
   356→function buildOptions(
   357→  toolId: string,
   358→  opts: Options
   359→): Record<string, unknown> {
   360→  switch (toolId) {
   361→    case "split":
   362→      return {
   363→        mode: opts.splitMode,
   364→        ranges: opts.ranges,
   365→        everyN: opts.everyN,
   366→        selectedPages: [...opts.splitSelected],
   367→      };
   368→    case "compress":
   369→      return { level: opts.level };
   370→    case "rotate":
   371→      return { angle: opts.angle };
   372→    case "pdf-to-jpg":
   373→      return { scale: opts.jpgScale };
   374→    case "jpg-to-pdf":
   375→      return {
   376→        pageSize: opts.pageSize,
   377→        margin: opts.margin,
   378→        orientation: opts.orientation,
   379→      };
   380→    case "word-to-pdf":
   381→      return { pageSize: opts.docPageSize };
   382→    case "watermark":
   383→      return {
   384→        text: opts.wmText,
   385→        fontSize: opts.wmSize,
   386→        color: opts.wmColor,
   387→        opacity: opts.wmOpacity,
   388→        rotation: opts.wmRotation,
   389→        position: opts.wmPosition,
   390→        bold: opts.wmBold,
   391→      };
   392→    case "page-numbers":
   393→      return {
   394→        format: opts.pnFormat,
   395→        position: opts.pnPosition,
   396→        startAt: opts.pnStart,
   397→        fontSize: opts.pnSize,
   398→        skipFirst: opts.pnSkipFirst,
   399→      };
   400→    case "crop":
   401→      return {
   402→        top: opts.cropTop,
   403→        right: opts.cropRight,
   404→        bottom: opts.cropBottom,
   405→        left: opts.cropLeft,
   406→      };
   407→    case "protect":
   408→      return { password: opts.password, confirm: opts.confirm };
   409→    case "unlock":
   410→      return { password: opts.password };
   411→    case "organize":
   412→      return {
   413→        order: opts.order.filter((p) => !opts.organizeDeleted.has(p)),
   414→        rotations: opts.rotations,
   415→      };
   416→    default:
   417→      return {};
   418→  }
   419→}
   420→
   421→function hint(toolId: string, opts: Options, files: StoreFile[]): string {
   422→  const tool = toolById(toolId);
   423→  if (!tool) return "";
   424→  if (files.length < tool.minFiles)
   425→    return `Add ${tool.minFiles === 1 ? "a file" : `at least ${tool.minFiles} files`} to continue`;
   426→  if (toolId === "watermark" && !opts.wmText.trim()) return "Enter watermark text";
   427→  if (toolId === "protect" && opts.password.length < 4)
   428→    return "Password needs at least 4 characters";
   429→  if (toolId === "protect" && opts.password !== opts.confirm)
   430→    return "Passwords do not match";
   431→  if (toolId === "split") {
   432→    if (opts.splitMode === "ranges" && !opts.ranges.trim())
   433→      return "Enter page ranges like 1-3, 5";
   434→    if (opts.splitMode === "extract" && opts.splitSelected.size === 0)
   435→      return "Select at least one page below";
   436→  }
   437→  return "";
   438→}

   981→function WatermarkOverlay({ opts }: { opts: Options }) {
   982→  const style: React.CSSProperties = {
   983→    color: opts.wmColor,
   984→    opacity: opts.wmOpacity,
   985→    fontSize: `${Math.max(8, opts.wmSize / 4)}px`,
   986→    fontWeight: opts.wmBold ? 800 : 400,
   987→    transform: `rotate(${opts.wmPosition === "header" || opts.wmPosition === "footer" || opts.wmPosition === "center" ? 0 : opts.wmRotation}deg)`,
   988→    letterSpacing: "0.02em",
   989→    whiteSpace: "nowrap",
   990→    pointerEvents: "none",
   991→  };
   992→  if (opts.wmPosition === "tile") {
   993→    return (
   994→      <div className="absolute inset-0 overflow-hidden">
   995→        {[15, 45, 75].map((t) => (
   996→          <span
   997→            key={t}
   998→            style={{ ...style, position: "absolute", left: `${t - 20}%`, top: `${t}%` }}
   999→          >
  1000→            {opts.wmText}
  1001→          </span>
  1002→        ))}
  1003→      </div>
  1004→    );
  1005→  }
  1006→  if (opts.wmPosition === "footer")
  1007→    return (
  1008→      <span style={{ ...style, position: "absolute", left: "50%", bottom: "6%", transform: "translateX(-50%)" }}>
  1009→        {opts.wmText}
  1010→      </span>
  1011→    );
  1012→  if (opts.wmPosition === "header")
  1013→    return (
  1014→      <span style={{ ...style, position: "absolute", left: "50%", top: "6%", transform: "translateX(-50%)" }}>
  1015→        {opts.wmText}
  1016→      </span>
  1017→    );
  1018→  if (opts.wmPosition === "center")
  1019→    return (
  1020→      <span style={{ ...style, position: "absolute", left: "50%", top: "50%", transform: "translate(-50%,-50%)" }}>
  1021→        {opts.wmText}
  1022→      </span>
  1023→    );
  1024→  return (
  1025→    <span
  1026→      style={{ ...style, position: "absolute", left: "18%", top: "46%" }}
  1027→    >
  1028→      {opts.wmText}
  1029→    </span>
  1030→  );
  1031→}
